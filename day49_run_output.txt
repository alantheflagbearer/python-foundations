class AlexNetStyle(nn.Module):
    def __init__(self, n_classes=4, p_drop=0.5):
        super().__init__()
        self.conv1 = nn.Conv2d(3, 4, kernel_size=3)
        self.conv2 = nn.Conv2d(4, 8, kernel_size=3)
        self.conv3 = nn.Conv2d(8, 16, kernel_size=3)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc1 = nn.Linear(16 * 3 * 3, 32)
        self.fc2 = nn.Linear(32, 16)
        self.fc3 = nn.Linear(16, n_classes)
        self.drop = nn.Dropout(p=p_drop)   # p is DROP prob = 1 - keep_prob

    def forward(self, x):
        x = self.pool(torch.relu(self.conv1(x)))
        x = self.pool(torch.relu(self.conv2(x)))
        x = self.pool(torch.relu(self.conv3(x)))
        x = x.flatten(1)
        x = self.drop(torch.relu(self.fc1(x)))
        x = self.drop(torch.relu(self.fc2(x)))
        return self.fc3(x)                # CrossEntropyLoss applies softmax

nn.Dropout does exactly what dropout_forward/dropout_backward do here -- zero a random subset and rescale survivors by 1/keep_prob during training, pass through unchanged during eval -- with autograd handling the backward pass instead of it being written out by hand.
