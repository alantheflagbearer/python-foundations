4. WHAT THIS MAPS TO IN PYTORCH

class ResNetBasicBlock(nn.Module):
    def __init__(self, channels):
        super().__init__()
        self.conv1 = nn.Conv2d(channels, channels, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm2d(channels)
        self.conv2 = nn.Conv2d(channels, channels, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm2d(channels)

    def forward(self, x):
        out = torch.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))     # no relu yet -- matches resblock_bn_forward's bn2
        return torch.relu(out + x)           # add the shortcut, THEN relu

# this IS the real ResNet (He et al., 2015) basic block, exactly as
# published -- today's ResNetBNConvNet is the closest this series gets
# to reproducing it. nn.BatchNorm2d also tracks running mean/variance
# for use at inference time on a SINGLE example -- today's spatial_
# batchnorm_forward always uses the current batch's own statistics,
# matching Day 40's simplification (fine here since this series' own
# accuracy() always evaluates a full array at once, never one example).

nn.BatchNorm2d(channels) tracks a running mean/variance during training (a detail this lesson's spatial_batchnorm_forward skips, following Day 40's convention) so it can normalize a single example at inference time without needing a batch of its own -- otherwise, conv -> BN -> ReLU, twice, then an identity add before the final ReLU, is exactly resblock_bn_forward's structure above.
