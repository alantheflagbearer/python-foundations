class JunctionImageDataset(torch.utils.data.Dataset):
    def __init__(self, file_paths, labels):
        self.file_paths, self.labels = file_paths, labels

    def __len__(self):
        return len(self.labels)

    def __getitem__(self, i):              # called lazily, per-index, by the DataLoader
        return np.load(self.file_paths[i]), self.labels[i]

loader = torch.utils.data.DataLoader(
    JunctionImageDataset(file_paths, labels),
    batch_size=20, shuffle=True,            # replaces this lesson's rng.permutation() call
    num_workers=4, prefetch_factor=2,       # replaces PrefetchDiskLoader's ONE background thread
    pin_memory=True)                        # a GPU-specific optimization with no equivalent in this lesson

__getitem__ is exactly LazyDiskLoader's _read_batch, called once per image instead of once per batch. num_workers>1 spreads PrefetchDiskLoader's single background thread across multiple PROCESSES (sidestepping Python's GIL, unlike this lesson's ThreadPoolExecutor), and prefetch_factor controls how many batches ahead each worker reads, generalizing this lesson's one-batch-ahead prefetch to N batches ahead.
